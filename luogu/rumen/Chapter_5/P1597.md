{

    for(int i=1;i+4<=strlen(zifuchuan+1);i+=5)
    {
        char c=0;
        c=zifuchuan[i];
        int num=0;
        num=abc(c);

        if(isdigit(zifuchuan[i+3]))
        {
            value[num]=zifuchuan[i+3]-48;
        }

        else
        {
            value[num]=value[abc(zifuchuan[i+3])];
        }
    }

}
