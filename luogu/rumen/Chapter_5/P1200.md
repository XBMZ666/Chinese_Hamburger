#include <stdio.h>

int main()
{
    char HUIXING[10]={0};
    char DUIWU[10]={0};

    scanf("%s",HUIXING+1);
    scanf("%s",DUIWU+1);

    int chengjiA=1;
    for(int i=1;HUIXING[i]!=0;i++)
    {
        int n=0;
        n=HUIXING[i]-64;
        chengjiA*=n;
    }

    int chengjiB=1;
    for(int j=1;DUIWU[j]!=0;j++)
    {
        int n=0;
        n=DUIWU[j]-64;
        chengjiB*=n;
    }

    if((chengjiA%47)==(chengjiB%47))
    {
        printf("GO");
    }
    else
    {
        printf("STAY");
    }
    return 0;
}
