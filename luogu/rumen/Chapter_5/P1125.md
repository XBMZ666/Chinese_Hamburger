#include <stdio.h>

int main()
{
    int arr[27]={0};

    char word[120]={0};

    scanf("%s",word+1);

    for(int i=1;word[i]!='\0';i++)
    {
        arr[(word[i]-96)]++;
    }

    int max=0;
    int min=0;
    
    for(int j=1;j<=26;j++)
    {
        if(j==1)
        {
            max=arr[j];
        }
        if(arr[j]>=max)
        {
            max=arr[j];
        }
    }

    int first=0;
    for(int k=1;k<=26;k++)
    {
        if(first==0&&arr[k]!=0)
        {
            min=arr[k];
            first=1;
        }
        if(min>=arr[k]&&arr[k]!=0)
        {
            min=arr[k];
        }
    }

    int value=0;
    value=max-min;

    int flag=1;
    if(value==1||value==0)
    {
        flag=0;
    }
    for(int m=2;m<=value-1;m++)
    {
        if(value%m==0)
        {
            flag=0;
            break;
        }
    }

    if(flag)
    {
        printf("Lucky Word\n");
        printf("%d",value);
    }
    else
    {
        printf("No Answer\n");
        printf("%d",0);
    }
    return 0;
}
