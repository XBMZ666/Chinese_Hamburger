#include <stdio.h>
#include <string.h>
#include <stdlib.h>

int main()
{
    int i=0;
    scanf("%d",&i);

    char op=0;
    for(int k=1;k<=i;k++)
    {
        int x=0;
        int y=0;

        char first[30]={0};

        scanf("%s",first);

        if(first[0]=='a'||first[0]=='b'||first[0]=='c')
        {
            op=first[0];
            scanf("%d %d",&x,&y);
        }

        else
        {
            x=atoi(first);
            scanf("%d",&y);
        }

        int result=0;
        char sign=0;

        
        if(op=='a')
        {
            result=x+y;
            sign='+';
        }

        else if(op=='b')
        {
            result=x-y;
            sign='-';
        }

        else if(op=='c')
        {
            result=x*y;
            sign='*';
        }
        
        char zifuchuan[100]={0};
        sprintf(zifuchuan,"%d%c%d=%d",x,sign,y,result);

        printf("%s\n%d\n",zifuchuan,strlen(zifuchuan));
        
    }
    
    return 0;
}
