#include <stdio.h>

int main()
{
    int n=0;
    scanf("%d",&n);

    char arr[200]={0};
    
    scanf("%s",arr+1);

    int sum=0;
    for(int i=1;i<=n;i++)
    {
        if(arr[i]=='V'&&arr[i+1]=='K')
        {
            arr[i]='P';
            arr[i+1]='P';
            sum++;
        }
    }

    for(int j=1;j<=n;j++)
    {
        if(arr[j]=='V'&&arr[j+1]=='V')
        {
            sum++;
            break;
        }
        if(arr[j]=='K'&&arr[j+1]=='K')
        {
            sum++;
            break;
        }
    }

    printf("%d",sum);
    return 0;
}
