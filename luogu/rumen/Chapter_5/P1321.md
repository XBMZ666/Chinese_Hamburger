```c
#include <stdio.h>

int main()
{
    char arr[300]={0};
    
    scanf("%s",arr);

    int suma=0;
    int sumb=0;
    for(int i=0;arr[i]!='\0';i++)
    {
        if(arr[i]=='b'||arr[i+1]=='o'||arr[i+2]=='y')
        {
            suma++;
        }
        if(arr[i]=='g'||arr[i+1]=='i'||arr[i+2]=='r'||arr[i+3]=='l')
        {
            sumb++;
        }
    }

    printf("%d\n%d\n",suma,sumb);
    return 0;
}
```
