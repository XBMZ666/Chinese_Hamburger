```c
#include <stdio.h>
int main()
{
    int k=0;
    scanf("%d",&k);

    int gold=0;
    int day=1;
    for(int i=1;;i++)
    {
        for(int j=1;j<=i;j++)
        {
            if(day>k)
            {
                break;
            }
            gold+=i;
            day++;
        }
        if(day>k)
        {
            break;
        }
    }

    printf("%d",gold);
    return 0;
}
```
