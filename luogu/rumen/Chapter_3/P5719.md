#include <stdio.h>
int main()
{
    int n=0;
    int k=0;
    scanf("%d %d",&n,&k);

    int A=0;
    int B=0;
    int suma=0;
    int sumb=0;

    for(int i=1;i<=n;i++)
    {
        if(i%k==0)
        {
            A+=i;
            suma+=1;
        }
        else if(i%k!=0)
        {
            B+=i;
            sumb+=1;
        }
    }

    double avea=0;
    double aveb=0;
    avea=(A*1.0)/suma;
    aveb=(B*1.0)/sumb;

    printf("%.1f %.1f",avea,aveb);
    return 0;
}
