int main()

        scanf("%d",arr+i);
    }

    int min=0;

    for(int j=1;j<=n;j++)
    {
        if(j==1)
        {
            max=arr[j];
        }
        if(arr[j]>=max)
        {
            max=arr[j];
        }
    }

    for(int k=1;k<=n;k++)
    {
        if(k==1)
        {
            min=arr[k];
        }
        if(min>=arr[k])
        {
            min=arr[k];
        }
    }
