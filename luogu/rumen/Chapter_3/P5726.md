#include <stdio.h>

int main()
{
    int n=0;
    scanf("%d",&n);

    int score[1100]={0};

    for(int i=1;i<=n;i++)
    {
        scanf("%d",score+i);
    }

    int min=0;
    int max=0;

    for(int k=1;k<=n;k++)
    {
        if(k==1)
        {
            min=score[k];
        }
        if(min>=score[k])
        {
            min=score[k];
        }
    }

    for(int m=1;m<=n;m++)
    {
        if(m==1)
        {
            max=score[m];
        }
        if(score[m]>=max)
        {
            max=score[m];
        }
    }

    for(int i=1;i<=n;i++)
    {
        if(score[i]==min)
        {
            score[i]=0;
            break;
        }
    }

    for(int j=1;j<=n;j++)
    {
        if(score[j]==max)
        {
            score[j]=0;
            break;
        }
    }

    int sum=0;
    for(int p=1;p<=n;p++)
    {
        sum+=score[p];
    }

    double average=0;
    average=(sum*1.0)/(n-2);

    printf("%.2f",average);

    
    return 0;
}
