#include <stdio.h>
int main()
{
    int n=0;
    int x=0;
    scanf("%d %d",&n,&x);

    int sum=0;
    for(int i=1;i<=n;i++)
    {
        int m=1;
        int value=i;
        while(i>=10)
        {
            i/=10;
            m++;
        }
        i=value;
        for(int j=1;j<=m;j++)
        {
            if(i%10==x)
                sum++;
            i/=10;
        }
        i=value;
    }
    printf("%d",sum);
    return 0;
}
