#include <stdio.h>
int main()
{
    int L=0;
    scanf("%d",&L);

    int sum=0;
    int num=0;
    for(int i=2;;i++)
    {
        int flag=1;
        for(int j=2;j<=i-1;j++)
        {
            if(i%j==0)
            {
                flag=0;
                break;
            }
        }
        if(flag)
        {
            sum+=i;
            if(sum<=L)
            {
                printf("%d\n",i);
                num++;
            }
            else
            {
                break;
            }
        }
        
    }
    printf("%d\n",num);
    return 0;
}
