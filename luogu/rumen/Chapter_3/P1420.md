int main()
{

    scanf("%d",&n);

    for(int i=1;i<=n;i++)
    {
        scanf("%d",arr+i);
    }

    int max=0;
    int lianxu=1;
    for(int i=1;i<=n;i++)
    {
        if(i==1)
        {
            max=lianxu;
        }
        
        if((arr[i+1])==(arr[i]+1))
        {
            lianxu++;
        }
        
        else if((arr[i+1])!=(arr[i]+1))
        {
            if(lianxu>=max)
            {
                max=lianxu;
            }
            lianxu=1;
        }
    }

    printf("%d",max);
    return 0;
}
