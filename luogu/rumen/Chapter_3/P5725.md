```c
#include <stdio.h>

int main()
{
    int N=0;
    scanf("%d",&N);

    int suma=1;
    int sumb=1;

    for(int i=1;i<=N;i++)
    {
        for(int j=1;j<=N;j++)
        {
            printf("%02d",suma);
            suma++;
        }
        printf("\n");
    }

    printf("\n");

    for(int k=1;k<=N;k++)
    {
        for(int p=1;p<=(N-k)*2;p++)
        {
            printf(" ");
        }
        for(int m=1;m<=k;m++)
        {
            printf("%02d",sumb);
            sumb++;
        }
        printf("\n");
    }
    return 0;
}
```
