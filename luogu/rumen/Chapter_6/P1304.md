#include <stdio.h>
#include <math.h>
int zhishu(int n)
{
    int flag=1;
    if(n==0||n==1)
    {
        flag=0;
        return flag;
    }

    if(n%2==0&&n!=2)
    {
        flag=0;
        return flag;
    }

    for(int i=3;i<=sqrt(n);i+=2)
    {
        if(n%i==0)
        {
            flag=0;
            break;
        }
    }
    
    return flag;
}

int main()
{
    int N=0;
    scanf("%d",&N);

    for(int i=4;i<=N;i+=2)
    {
        int kefen=0;
        for(int j=1;j<=i;j++)
        {
            if(zhishu(j))
            {
                if(zhishu(i-j))
                {
                    printf("%d=%d+%d\n",i,j,i-j);
                    kefen=1;
                    break;
                }
            }
            if(kefen)
            {
                break;
            }
        }

    }
    return 0;
}
