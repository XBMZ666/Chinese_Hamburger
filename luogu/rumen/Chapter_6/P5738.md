#include <stdio.h>
double degrade(int* first,int m)
{
    int max=0;
    for(int i=0;i<=m-1;i++)
    {
        if(i==0)
        {
            max=*(first+i);
        }
        if(*(first+i)>=max)
        {
            max=*(first+i);
        }
    }

    int min=0;
    for(int j=0;j<=m-1;j++)
    {
        if(j==0)
        {
            min=*(first+j);
        }
        if(*(first+j)<=min)
        {
            min=*(first+j);
        }
    }

    for(int i=0;i<=m-1;i++)
    {
        if(*(first+i)==max)
        {
            *(first+i)=0;
            break;
        }
    }

    for(int j=0;j<=m-1;j++)
    {
        if(*(first+j)==min)
        {
            *(first+j)=0;
            break;
        }
    }

    int sum=0;
    for(int k=0;k<=m-1;k++)
    {
        sum+=*(first+k);
    }

    double average=0;
    average=(sum*1.0)/(m-2);
    

    return average;
}

int main()
{
    int n=0;
    int m=0;
    scanf("%d %d",&n,&m);

    double MAX=0;
    for(int p=0;p<n;p++)
    {
        double grade=0;
        int arr[30]={0};
        for(int q=0;q<m;q++)
        {
            scanf("%d",arr+q);
        }

        grade=degrade(arr,m);
        if(p==0)
            MAX=grade;
        if(grade>=MAX)
            MAX=grade;
    }
    
    printf("%.2f",MAX);
    return 0;
}
