#include <stdio.h>

struct XUEYUAN
{
    char xingming[18];
    int nianling;
    int chengji;
};

int main()
{
    int n=0;
    scanf("%d",&n);

    struct XUEYUAN arr[8]={0};
    
    for(int i=1;i<=n;i++)
    {
        struct XUEYUAN student;
        scanf("%s %d %d",student.xingming,&student.nianling,&student.chengji);
        arr[i]=student;
    }

    for(int j=1;j<=n;j++)
    {
        double NOI=0;
        int NOINOI=0;
        NOI=(arr[j].chengji)*1.2;
        if(NOI>=600)
        {
            NOI=600;
        }
        NOINOI=(int)NOI;
        
        printf("%s %d %d\n",arr[j].xingming,arr[j].nianling+1,NOINOI);
    }
    
    return 0;
}
