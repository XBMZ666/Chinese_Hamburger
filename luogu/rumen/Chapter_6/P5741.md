#include <stdio.h>
#include <math.h>

struct student
{
    char xingming[10];
    int yuwen;
    int shuxue;
    int yingyu;
    int zongfen;
};

int main()
{
    int N=0;
    scanf("%d",&N);

    struct student arr[1200]={0};

    for(int i=1;i<=N;i++)
    {
        struct student stu;
        scanf("%s %d %d %d",stu.xingming,&stu.yuwen,&stu.shuxue,&stu.yingyu);
        stu.zongfen=stu.yuwen+stu.shuxue+stu.yingyu;
        arr[i]=stu;
    }

    for(int j=1;j<=N-1;j++)
    {
        for(int k=1;k<=N-j;k++)
        {
            int flag=1;
            if((fabs(arr[j].yuwen-arr[j+k].yuwen))>5)
            {
                flag=0;
            }
            if((fabs(arr[j].shuxue-arr[j+k].shuxue))>5)
            {
                flag=0;
            }
            if((fabs(arr[j].yingyu-arr[j+k].yingyu))>5)
            {
                flag=0;
            }
            if((fabs(arr[j].zongfen-arr[j+k].zongfen))>10)
            {
                flag=0;
            }
            if(flag)
            {
                printf("%s %s\n",arr[j].xingming,arr[j+k].xingming);
            }
        }
    }
    return 0;
}
