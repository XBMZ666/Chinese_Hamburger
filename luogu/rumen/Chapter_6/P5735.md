#include <stdio.h>
#include <math.h>
typedef double d;
d distance(d x1,d y1,d x2,d y2)
{
    d m;
    m=sqrt(pow(x2-x1,2)+pow(y2-y1,2));
    return m;
}

int main()
{
    d x1,y1,x2,y2,x3,y3;
    d C=0;

    scanf("%lf%lf%lf%lf%lf%lf",&x1,&y1,&x2,&y2,&x3,&y3);
    C=distance(x1,y1,x2,y2)+distance(x2,y2,x3,y3)+distance(x3,y3,x1,y1);

}
