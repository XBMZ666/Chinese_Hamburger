#include <stdio.h>
int zhishu(int x)
{
    int flag=1;
    if(x==1)
        flag=0;
    for(int k=2;k<=x-1;k++)
    {
        if(x%k==0)
        {
            flag=0;
            break;
        }
    }
    return flag;
}
int main()
{
    int n=0;
    scanf("%d",&n);
    
    int arr[200]={0};
    
    for(int i=1;i<=n;i++)
        scanf("%d",arr+i);

    for(int j=1;j<=n;j++)
    {
        if((zhishu(arr[j]))!=1)
            arr[j]=0;
    }

    for(int m=1;m<=n;m++)
    {
        if(arr[m]==0)
        {
            continue;
        }
        printf("%d ",arr[m]);
    }
    return 0;
}
