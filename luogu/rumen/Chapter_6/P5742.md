#include <stdio.h>

struct student
{
  int xuehao;
  int xueye;
  int suzhi;
  int zonghe;
};

int youxiu(struct student x)
{
    int flaga=0;
    int flagb=0;

    if(x.xueye+x.suzhi>140)
    {
        flaga=1;
    }

    if(x.zonghe>=800)
    {
        flagb=1;
    }
    
    return (flaga&&flagb);
}

int main()
{
    int N=0;
    scanf("%d",&N);

    struct student arr[1200]={0};
    
    for(int i=1;i<=N;i++)
    {
        struct student stu={0};
        scanf("%d %d %d",&stu.xuehao,&stu.xueye,&stu.suzhi);
        stu.zonghe=stu.xueye*7+stu.suzhi*3;
        arr[i]=stu;
    }

    for(int j=1;j<=N;j++)
    {
        if(youxiu(arr[j]))
        {
            printf("Excellent\n");
        }
        else
        {
            printf("Not excellent\n");
        }
    }
    return 0;
}
