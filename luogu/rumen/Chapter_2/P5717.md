    if(a+b<=c||fabs(a-b)>=c)
        printf("Not triangle");
    else
    {
        double cosa=0;
        cosa=((pow(a,2)+pow(b,2)-pow(c,2))/(2*a*b));
        if(cosa==0)
            printf("Right triangle\n");
        else if(cosa>0)
            printf("Acute triangle\n");
        else
            printf("Obtuse triangle\n");

                printf("Equilateral triangle\n");
        }
    }

}
