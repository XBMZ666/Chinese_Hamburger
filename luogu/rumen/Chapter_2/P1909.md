#include <stdio.h>
#include <math.h>
int main()
{
    int n=0;

    int pricec=0;

    int a=0;
    int anum=0;
    int b=0;
    int bnum=0;
    int c=0;
    int cnum=0;

    scanf("%d %d",&anum,&a);
    pricea=(int)(ceil((n*1.0)/anum))*a;

    scanf("%d %d",&bnum,&b);
    priceb=(int)(ceil((n*1.0)/bnum))*b;
    
    scanf("%d %d",&cnum,&c);
    pricec=(int)(ceil((n*1.0)/cnum))*c;


    int min=pricea;

        min=priceb;
    if(pricec<=min)
        min=pricec;

}
