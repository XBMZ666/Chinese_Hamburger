#include <stdio.h>
int main()
{
    int arr[8]={0};
    int flag=0;
    for(int k=1;k<=7;k++)
    {

        int b=0;
        scanf("%d %d",&a,&b);
        arr[k]=a+b;
        if(a+b>8)
            flag=1;
    }

    if(!flag)
        printf("%d",0);
    else
    {
        int value=arr[1];
        for(int i=1;i<=7;i++)
        {
            if(arr[i]>=value)
                value=arr[i];
        }

        for(int j=1;j<=7;j++)
        {
            if(arr[j]==value)
            {
                printf("%d",j);
                break;
            }
        }
    }
    
    
    return 0;
}
