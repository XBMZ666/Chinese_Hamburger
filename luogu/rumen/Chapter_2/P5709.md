#include <stdio.h>

int main()
{
    int m=0;
    int t=0;
    int s=0;
    scanf("%d %d %d",&m,&t,&s);

    if(t==0)
        printf("%d",0);

    else
    {
        if(s<t)
        {
            printf("%d",m-1);
        }
        else
        {
            if(s/t>=m)
            {
                printf("%d",0);
            }
            else
            {
                if(s%t==0)
                {
                    printf("%d",m-s/t);
                }
                else
                {
                    printf("%d",m-s/t-1);
                }
            }
        }
    }
    return 0;
}
