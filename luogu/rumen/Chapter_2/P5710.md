#include <stdio.h>
int main()
{
    int arr[4]={0};
    int a=0;
    int b=0;

    int x=0;
    scanf("%d",&x);
    
    if(x%2==0)
        a=1;
    if(x>4&&x<=12)
        b=1;

    if(a&&b)
        arr[0]=1;
    if(a||b)
        arr[1]=1;
    if(a&&!b)
        arr[2]=1;
    if(!a&&b)
        arr[2]=1;
    if(!a&&!b)
        arr[3]=1;

    for(int i=0;i<4;i++)
        printf("%d ",arr[i]);
    return 0;
}
