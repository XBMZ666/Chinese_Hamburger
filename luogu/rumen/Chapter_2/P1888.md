#include <stdio.h>
int ouji(int a,int b)
{
    while(b!=0)
    {
        int value=a%b;
        a=b;
        b=value;
    }
    return a;
}

void easy(int a,int b)
{
    int yueshu=ouji(a,b);
    printf("%d/%d",a/yueshu,b/yueshu);
}
int main()
{
    int a=0;
    int b=0;
    int c=0;
    scanf("%d %d %d",&a,&b,&c);

    int min=a;
    if(b<=min)
        min=b;
    if(c<=min)
        min=c;

    int max=a;
    if(b>=max)
        max=b;
    if(c>=max)
        max=c;

    easy(min,max);
    return 0;
}
