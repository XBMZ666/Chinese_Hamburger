#include <stdio.h>

int main()
{
    int n=0;
    scanf("%d",&n);

    int zhongjiang[8]={0};
    int qingkuang[8]={0};

    for(int i=1;i<=7;i++)
    {
        scanf("%d",zhongjiang+i);
    }
    
    for(int i=1;i<=n;i++)
    {
        int linshi[8];
        scanf("%d %d %d %d %d %d %d",linshi+1,linshi+2,linshi+3,linshi+4,linshi+5,linshi+6,linshi+7);

        int num=0;
        
        for(int j=1;j<=7;j++)
        {
            for(int k=1;k<=7;k++)
            {
                if(linshi[j]==zhongjiang[k])
                {
                    num++;
                }
            }
        }

        qingkuang[8-num]++;
        
    }

    printf("%d %d %d %d %d %d %d",qingkuang[1],qingkuang[2],qingkuang[3],qingkuang[4],qingkuang[5],qingkuang[6],qingkuang[7]);    
    return 0;
}
