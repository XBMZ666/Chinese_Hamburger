```c
#include <stdio.h>

int main()
{
    int N=0;
    scanf("%d",&N);

    int n=0;
    int k=0;
    int sum=0;
    while((scanf("%d",&n))!=EOF)
    {
        for(int i=1;i<=n;i++)
        {
            printf("%d",k);
            sum++;
            if(sum==N)
            {
                printf("\n");
                sum=0;
            }
        }

        if(k==0)
        {
            k=1;
        }
        else
        {
            k=0;
        }
    }
}
```
