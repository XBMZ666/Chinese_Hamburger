#include <stdio.h>

int main()
{
    int Sa=0;
    int Sb=0;
    int Sc=0;

    scanf("%d %d %d",&Sa,&Sb,&Sc);
    
    int count[85]={0};

    for(int i=1;i<=Sa;i++)
    {
        for(int j=1;j<=Sb;j++)
        {
            for(int k=1;k<=Sc;k++)
            {
                count[i+j+k]++;
            }
        }
    }

    int max=0;
    for(int m=0;m<85;m++)
    {
        if(m==0)
        {
            max=count[m];
        }
        if(count[m]>=max)
        {
            max=count[m];
        }
    }

    for(int u=0;u<85;u++)
    {
        if(max==count[u])
        {
            printf("%d",u);
            break;
        }
    }
    return 0;
}
