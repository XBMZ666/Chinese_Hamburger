#include <math.h>

    for(int i=1;i<=n;i++)
    {
        double a=0;
        int t=0;
        scanf("%lf %d",&a,&t);
        for(int j=1;j<=t;j++)
        {
            int quzheng=0;
            quzheng=(floor)(j*a);
            arr[quzheng]=(!arr[quzheng]);
        }
    }

    for(int m=1;;m++)
    {
        if(arr[m]==1)
        {
            printf("%d",m);
            break;
        }
    }
    return 0;
}
