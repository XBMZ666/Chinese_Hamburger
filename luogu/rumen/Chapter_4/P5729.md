#include <stdio.h>

int main()
{
    int chang=0;
    int kuan=0;
    int gao=0;
    scanf("%d %d %d",&chang,&kuan,&gao);

    int arr[30][30][30]={0};

    
    int q=0;
    scanf("%d",&q);

    for(int m=1;m<=q;m++)
    {
        int x=0;
        int y=0;
        int z=0;
        int a=0;
        int b=0;
        int c=0;
        scanf("%d %d %d %d %d %d",&x,&y,&z,&a,&b,&c);
        for(int i=x;i<=a;i++)
        {
            for(int j=y;j<=b;j++)
            {
                for(int k=z;k<=c;k++)
                {
                    if(arr[i][j][k]==0)
                    {
                        arr[i][j][k]=1;
                    }
                }
            }
        }
    }

    int sum=0;

    for(int i=1;i<=chang;i++)
    {
        for(int j=1;j<=kuan;j++)
        {
            for(int k=1;k<=gao;k++)
            {
                if(arr[i][j][k]==0)
                {
                    sum++;
                }
            }
        }
    }


    printf("%d",sum);
    return 0;
}
