```c
#include <stdio.h>
int main()
{
    int arr[150]={0};

    int num=0;
    int i=0;
    while(1)
    {
        scanf("%d",&num);
        if(num==0)
        {
            break;
        }
        i++;
        arr[i]=num;
    }

    for(int j=i;j>=1;j--)
    {
        printf("%d ",arr[j]);
    }
    return 0;
}
```
