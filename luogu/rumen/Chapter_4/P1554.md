#include <stdio.h>
int main()
{
    int arr[10]={0};

    int M=0;
    int N=0;
    scanf("%d %d",&M,&N);
    for(int i=M;i<=N;i++)
    {
        int value=i;
        while(i>0)
        {
            for(int j=0;j<=9;j++)
            {
                if(i%10==j)
                {
                    arr[j]+=1;
                    break;
                }
            }
            i/=10;
        }
        i=value;
    }

    for(int k=0;k<=9;k++)
    {
        printf("%d ",arr[k]);
    }
}
