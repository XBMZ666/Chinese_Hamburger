#include <stdio.h>
int main()
{
    int arr[200]={0};

    int n=0;
    scanf("%d",&n);

    int num=0;
    while(n!=1)
    {
        arr[num+1]=n;
        if(n%2!=0)
        {
            n=n*3+1;
        }
        else
        {
            n/=2;
        }
        num++;
    }

    printf("%d ",1);
    
    for(int j=num;j>=1;j--)
        printf("%d ",arr[j]);
    return 0;
}
