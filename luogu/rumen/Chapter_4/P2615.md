#include <stdio.h>

int main()
{
    int N=0;
    scanf("%d",&N);

    int arr[45][45]={0};

    int x=0;
    int y=0;
    for(int k=1;k<=(N*N);k++)
    {
        if(k==1)
        {
            arr[0][(N+1)/2-1]=k;
            x=1;
            y=(N+1)/2;
        }

        else if(x==1&&y!=N)
        {
            arr[N-1][y]=k;
            x=N;
            y=y+1;
        }

        else if(x!=1&&y==N)
        {
            arr[x-2][0]=k;
            x=x-1;
            y=1;
        }

        else if(x==1&&y==N)
        {
            arr[x][y-1]=k;
            x=x+1;
            y=y;
        }

        else if(x!=1&&y!=N)
        {
            if(arr[x-2][y]==0)
            {
                arr[x-2][y]=k;
                x=x-1;
                y=y+1;
            }
            else
            {
                arr[x][y-1]=k;
                x=x+1;
                y=y;
            }
        }
    }


    for(int i=0;i<=N-1;i++)
    {
        for(int j=0;j<=N-1;j++)
        {
            printf("%d ",arr[i][j]);
        }
        printf("\n");
    }
    return 0;
}
