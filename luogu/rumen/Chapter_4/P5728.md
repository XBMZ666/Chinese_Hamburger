#include <stdio.h>
#include <math.h>
int main()
{
    int yuwen[1200]={0};
    int shuxue[1200]={0};
    int yingyu[1200]={0};
    int zongfen[1200]={0};

    int N=0;
    scanf("%d",&N);
    
    for(int i=1;i<=N;i++)
    {
        scanf("%d %d %d",yuwen+i,shuxue+i,yingyu+i);
        zongfen[i]=yuwen[i]+shuxue[i]+yingyu[i];
    }

    int sum=0;

    for(int j=1;j<=N-1;j++)
    {
        for(int k=1;k<=N-j;k++)
        {
            if((fabs(yuwen[j]-yuwen[k+j])<=5)&&(fabs(shuxue[j]-shuxue[k+j])<=5)&&(fabs(yingyu[j]-yingyu[k+j])<=5)&&(fabs(zongfen[j]-zongfen[k+j])<=10))
                sum++;
        }
    }

    printf("%d",sum);
    return 0;
}
