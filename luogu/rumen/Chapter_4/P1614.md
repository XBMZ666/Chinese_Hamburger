```c
#include <stdio.h>

int main()
{
    int arr[5000]={0};

    int n=0;
    int m=0;
    scanf("%d %d",&n,&m);

    for(int i=1;i<=n;i++)
    {
        scanf("%d",arr+i);
    }

    int sum=0;
    for(int j=1;j<=m;j++)
    {
        sum+=arr[j];
    }

    int min=0;
    min=sum;
    for(int k=m+1;k<=n;k++)
    {
        sum=sum+arr[k]-arr[k-m];
        if(sum<=min)
        {
            min=sum;
        }
    }

    printf("%d",min);
    return 0;
}
```
