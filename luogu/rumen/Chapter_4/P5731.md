#include <stdio.h>

int main()
{
    int arr[12][12]={0};

    int n=0;
    scanf("%d",&n);

    int dx[4]={0,1,0,-1};
    int dy[4]={1,0,-1,0};

    int dir=0;
    int num=1;
    int x=0;
    int y=0;
    for(int i=0;i<(n*n);i++)
    {
        arr[x][y]=num;
        num++;

        int nx=0;
        int ny=0;
        nx=x+dx[dir];
        ny=y+dy[dir];

        if(nx<0||ny<0||nx>n-1||ny>n-1||arr[nx][ny]!=0)
        {
            dir=(dir+1)%4;
            nx=x+dx[dir];
            ny=y+dy[dir];
        }

        x=nx;
        y=ny;
    }

    for(int i=0;i<n;i++)
    {
        for(int j=0;j<n;j++)
        {
            printf("%3d",arr[i][j]);
        }
        printf("\n");
    }
    
    return 0;
}
