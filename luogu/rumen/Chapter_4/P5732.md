#include <stdio.h>
int main()
{
    int arr[25][25]={0};

    int n=0;
    scanf("%d",&n);
    
    for(int i=0;i<n;i++)
    {
        for(int j=0;j<i+1;j++)
        {
            if(j==0)
            {
                arr[i][j]=1;
            }
            else if(j==i)
            {
                arr[i][j]=1;
            }
            else
            {
                arr[i][j]=arr[i-1][j]+arr[i-1][j-1];
            }
        }
    }

    for(int i=0;i<n;i++)
    {
        for(int j=0;j<i+1;j++)
        {
            printf("%d ",arr[i][j]);
        }
        printf("\n");
    }
    return 0;
}
