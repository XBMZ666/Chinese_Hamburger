#include <stdio.h>
int main()
{
    int fish[150]={0};

    int n=0;
    scanf("%d",&n);

    for(int i=1;i<=n;i++)
    {
        scanf("%d",fish+i);
    }

    for(int i=1;i<=n;i++)
    {
        int sum=0;
        for(int j=1;j<i;j++)
        {
            if(fish[j]<fish[i])
            {
                sum++;
            }
        }
        printf("%d ",sum);
    }
    return 0;
}
