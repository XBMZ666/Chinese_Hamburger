#include <stdio.h>
int main()
{
    int tree[12000]={0};

    int l=0;
    int m=0;
    scanf("%d %d",&l,&m);

    for(int i=1;i<=m;i++)
    {
        int x=0;
        int y=0;
        scanf("%d %d",&x,&y);
        for(int k=x;k<=y;k++)
        {
            tree[k]=1;
        }
    }

    int sum=0;

    for(int j=0;j<=l;j++)
    {
        if(tree[j]==0)
            sum++;
    }

    printf("%d",sum);
    return 0;
}
