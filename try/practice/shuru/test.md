```c
#include <stdio.h>

//每次输入一串数据并回车后, 数据被存入缓冲区, scanf函数根据需求从缓冲区中读取数据, 如果未读取完则继续读取, 如果读取后数据过剩, 则留给之后的scanf函数
int main()
{
    int n = 0;

    printf("请输入n = ");
    scanf("%d", &n);

    printf("接下来请输入n个数： ");
    for (int i = 1; i <= n; i++)
    {
        int num = 0;
        scanf("%d", &num);

        if (i != 1)
        {
            printf(" ");
        }
        printf("%d", num);
    }

    printf("\n");

    int k = 0;
    scanf("%d", &k);
    printf("k = %d\n", k);

    return 0;
}

//以上代码, 如果输入n的值为4, 接下来输入1 2 3 4 5, 那么按下回车后, 这些数据全进入缓冲区
//接下来, 在for语句中的scanf函数会先读取1, 打印输出1, 再读取2, 打印输出2, 之后读取3, 打印输出3, 又读取4, 打印输出4
//再然后, 打印输出一个换行, 由于缓冲区还留有一个5, 所以接下来的scanf直接读走缓冲区的5, 而不会再在屏幕上给出输入提示, 之后打印输出k的值
//要是输入n的值为4, 且接下来输入1 2 3 4, 那么打印输出1 2 3 4后, 缓冲区无数据可读了, 屏幕会给出输入提示, 输入再回车后, 值在缓冲区被scanf读取, 最后打印输出k的值
```
