```c
#include <stdio.h>

int main()
{
    double s=0;
    scanf("%lf",&s);

    int step=1;
    double total=0;
    double distance=2;
    for(int i=1;;i++)
    {
        total+=distance;
        if(total>=s)
        {
            break;
        }
        distance*=0.98;
        step++;
    }
    
    printf("%d",step);
    return 0;
}
```
